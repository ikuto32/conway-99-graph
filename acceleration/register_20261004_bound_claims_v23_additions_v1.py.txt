def v23_descriptor(cid):
    if cid not in V23_EXACT:
        return None
    need(digest(ROOT/V23_DESCRIPTOR['path']) == V23_DESCRIPTOR['sha256'],
         'v23 exact eighteen descriptor bytes')
    declared = json.loads((ROOT/V23_DESCRIPTOR['path']).read_bytes())
    need(declared['schema'] == 'SOURCE_ONLY_WAVE45_EIGHTEEN_BOUND_CLAIM_DESCRIPTOR_V3'
         and type(declared['record_population']) is int and declared['record_population'] == 18
         and v15_same([r['id'] for r in declared['records']], list(V23_EXACT))
         and v15_same(declared['role_counts'], dict(new_exact_Native_verifier_ids=11,
             new_exact_ROOT_verifier_ids=2, ordinary_separate_verifier_roles=5,
             generic_Native_or_ROOT_waiver=False)), 'v23 exact population and role contract')
    rows = [r for r in declared['records'] if r['id'] == cid]
    need(len(rows) == 1, 'v23 one exact descriptor row')
    record = rows[0]
    identity, report_identity = V23_EXACT[cid]
    need(record['binding']['sha256'] == identity
         and record['bound_report']['declared_sha256'] == report_identity,
         'v23 static binding and report identities')
    return record


def v23_role(cid, expected, binding):
    if cid not in V23_EXACT:
        return False
    record = v23_descriptor(cid)
    need(expected == V23_EXACT[cid][0], 'v23 exact frozen binding identity')
    need(type(binding) is dict and {'id', 'revision', 'claim_revision', 'status',
         'review_state', 'kind', 'basis', 'statement', 'producer', 'verifier',
         'method', 'verification_timestamp', 'report', 'report_sha256', 'scope',
         'dependencies', 'target_resolution', 'assumptions', 'limitations'} <= set(binding), 'v23 required binding fields')
    need(type(binding['assumptions']) is list
         and all(type(item) is str for item in binding['assumptions']),
         'v23 explicit assumption metadata')
    need(type(binding['revision']) is int and type(binding['claim_revision']) is int
         and binding['revision'] == binding['claim_revision'] == 1,
         'v23 exact integer revision')
    for field in ('id', 'revision', 'status', 'review_state', 'kind', 'basis',
                  'statement', 'producer', 'verifier', 'method'):
        need(v15_same(binding[field], record[field]), 'v23 literal binding core '+field)
    need(binding['status'] == ('REFUTED' if cid == V23_REFUTED_ID else 'VERIFIED')
         and binding['review_state'] == 'CLEAR'
         and binding['producer'] != binding['verifier']
         and type(binding['verification_timestamp']) is str
         and binding['verification_timestamp'] == record['verification_timestamp_literal']
         and binding['report'] == record['bound_report']['path']
         and binding['report_sha256'] == V23_EXACT[cid][1]
         and binding.get('target_resolution') == 'NONE'
         and 'verification_records' not in binding, 'v23 separate exact roles and timestamp')
    need(v15_same(binding['scope'], record['original_scope'])
         and v15_same(binding['dependencies'], record['original_dependencies']),
         'v23 original scope and dependencies')
    need(digest(ROOT/record['binding']['path']) == expected, 'v23 original binding bytes')
    original = json.loads((ROOT/record['binding']['path']).read_bytes())
    need(v15_same(binding, original), 'v23 complete original typed binding')
    return True


def v23_verification_outcome(cid, expected, binding):
    if cid not in V23_EXACT:
        return None
    need(v23_role(cid, expected, binding), 'v23 exact status-specific role')
    declared = json.loads((ROOT/V23_DESCRIPTOR['path']).read_bytes())
    outcome = 'FAIL' if cid == V23_REFUTED_ID else 'PASS'
    need(v15_same(declared['status_semantics'][cid],
         dict(status=binding['status'], verification_outcome=outcome)),
         'v23 exact status-specific theorem outcome')
    if cid == V23_REFUTED_ID:
        contract = declared['original_generic_refutation_contract']
        need(binding['verification_outcome'] == 'FAIL'
             and binding['refutation']['exact_statement_disproved'] is True
             and binding['refutation']['proof'] == contract['proof']['path']
             and binding['refutation']['sha256'] == contract['proof']['sha256']
             and contract['id'] == cid and type(contract['revision']) is int
             and contract['revision'] == 1 and contract['binding_sha256'] == expected
             and contract['theorem_verification_outcome'] == 'FAIL'
             and contract['successful_counterexample_check_outcome'] == 'PASS',
             'v23 exact refutation evidence and FAIL interpretation')
    return outcome


def v23_scope(cid, expected, binding):
    need(v23_role(cid, expected, binding), 'v23 scoped exact binding')
    record = v23_descriptor(cid)
    original = binding['scope']
    text_key = 'description' if 'description' in original else 'population'
    flag_key = 'unrestricted_target' if 'unrestricted_target' in original else 'unrestricted'
    need(type(original[text_key]) is str and type(original[flag_key]) is bool
         and type(original['target_resolution']) is str
         and original['target_resolution'] == 'NONE', 'v23 typed scope projection fields')
    projected = dict(description=original[text_key], unrestricted_target=original[flag_key],
                     target_resolution=original['target_resolution'])
    need(v15_same(projected, record['proposed_three_key_scope']),
         'v23 exact three-key scope projection')
    return projected


def v23_dependencies(cid, expected, binding, claims):
    need(v23_role(cid, expected, binding), 'v23 dependency exact binding')
    result = []
    for original in binding['dependencies']:
        need(type(original) is dict and set(original) <= {'id', 'revision', 'relation', 'type', 'reason'}
             and ('relation' in original) != ('type' in original)
             and type(original['id']) is str and type(original['revision']) is int
             and original['revision'] == 1, 'v23 typed dependency projection')
        relation = original['relation'] if 'relation' in original else original['type']
        need(type(relation) is str and relation == 'uses_result', 'v23 exact dependency relation')
        previous = [c for c in claims if c['id'] == original['id']]
        need(len(previous) == 1 and type(previous[0]['revision']) is int
             and previous[0]['revision'] == original['revision']
             and previous[0]['status'] == 'VERIFIED' and previous[0]['review_state'] == 'CLEAR',
             'v23 preceding exact verified dependency revision')
        row = dict(id=original['id'], revision=original['revision'], relation=relation)
        if 'reason' in original:
            need(type(original['reason']) is str, 'v23 literal dependency reason')
            row['reason'] = original['reason']
        result.append(row)
    return result


def v23_evidence(cid):
    record = v23_descriptor(cid)
    need(record is not None, 'v23 exact evidence routing')
    declared = json.loads((ROOT/V23_DESCRIPTOR['path']).read_bytes())
    pins = {V23_DESCRIPTOR['path']: V23_DESCRIPTOR['sha256'],
            V23_CAPACITY_METADATA['path']: V23_CAPACITY_METADATA['sha256'],
            V23_SEVENTEEN_METADATA['path']: V23_SEVENTEEN_METADATA['sha256']}
    for review in declared['root_existing_metadata_reviews']:
        need(type(review) is dict and set(review) == {'path', 'sha256'}
             and type(review['path']) is str and type(review['sha256']) is str,
             'v23 typed exact Root review reference')
        need(review['path'] not in pins or pins[review['path']] == review['sha256'],
             'v23 consistent exact Root review identity')
        pins[review['path']] = review['sha256']
    proof = declared['original_generic_refutation_contract']['proof']
    need(type(proof) is dict and set(proof) == {'path', 'sha256'},
         'v23 exact refutation proof reference')
    need(proof['path'] not in pins or pins[proof['path']] == proof['sha256'],
         'v23 consistent refutation proof identity')
    pins[proof['path']] = proof['sha256']
    correction = declared['metadata_correction_provenance']['receipt']
    need(type(correction) is dict and correction == dict(
         path='acceleration/results/20261004_wave45_dual_gram_assumptions_metadata_correction01.json',
         sha256='fbc54e6cebdd86fdf2e881446739d55ef39ce3b1e55a551da085af97b5702530'),
         'v23 exact assumptions correction receipt')
    pins[correction['path']] = correction['sha256']
    for path, identity in pins.items():
        need(digest(ROOT/path) == identity, 'v23 exact descriptor and Root metadata evidence')
    return pins


def v23_report(cid, report_sha, binding, report):
    if cid not in V23_EXACT:
        return None
    need(v23_role(cid, V23_EXACT[cid][0], binding), 'v23 bound report exact role')
    record = v23_descriptor(cid)
    need(report_sha == V23_EXACT[cid][1] and digest(ROOT/binding['report']) == report_sha,
         'v23 exact original report bytes')
    original = json.loads((ROOT/binding['report']).read_bytes())
    need(v15_same(report, original) and report['status'] == record['bound_report']['headline']
         and type(report['statement']) is str and report['statement'] == binding['statement'],
         'v23 complete typed original report and literal statement')
    outcome = v23_verification_outcome(cid, V23_EXACT[cid][0], binding)
    if cid == V23_REFUTED_ID:
        need(report['outcome'] == 'FAIL' and report['counterexample_audit_outcome'] == 'PASS'
             and report['proof'] == binding['refutation']['proof']
             and report['proof_sha256'] == binding['refutation']['sha256'],
             'v23 successful counterexample check distinct from refuted theorem')
    v23_evidence(cid)
    return dict(adapter_version=23, descriptor=V23_DESCRIPTOR,
        recorded_binding=record['binding']['path'], recorded_binding_sha256=V23_EXACT[cid][0],
        recorded_report=binding['report'], recorded_report_sha256=report_sha,
        recorded_binding_statement=binding['statement'], raw_statement_changed=False,
        original_binding_scope=copy.deepcopy(binding['scope']),
        original_report_scope=copy.deepcopy(report.get('scope')),
        original_dependencies=copy.deepcopy(binding['dependencies']),
        schema_scope=v23_scope(cid, V23_EXACT[cid][0], binding),
        original_verification_timestamp=binding['verification_timestamp'],
        theorem_verification_outcome=outcome,
        successful_counterexample_audit_outcome=(report['counterexample_audit_outcome']
            if cid == V23_REFUTED_ID else None),
        no_new_mathematical_approval=True, generic_role_waiver=False,
        target_resolution='NONE')
